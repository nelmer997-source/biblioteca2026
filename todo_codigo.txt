package com.curso.biblioteca2026.config;

import org.springframework.boot.context.properties.EnableConfigurationProperties;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.oauth2.jose.jws.MacAlgorithm;
import org.springframework.security.oauth2.jwt.JwtDecoder;
import org.springframework.security.oauth2.jwt.JwtEncoder;
import org.springframework.security.oauth2.jwt.JwtValidators;
import org.springframework.security.oauth2.jwt.NimbusJwtDecoder;
import org.springframework.security.oauth2.jwt.NimbusJwtEncoder;

import javax.crypto.SecretKey;
import javax.crypto.spec.SecretKeySpec;
import java.time.Clock;
import java.util.Base64;

@Configuration
@EnableConfigurationProperties(JwtProperties.class)
public class JwtConfig {

    @Bean
    Clock clock() {
        return Clock.systemUTC();
    }

    @Bean
    SecretKey jwtSecretKey(JwtProperties properties) {
        if (properties.secret() == null || properties.secret().isBlank()) {
            throw new IllegalStateException("JWT_SECRET nÃ£o configurado");
        }
        byte[] chave;
        try {
            chave = Base64.getDecoder().decode(properties.secret().trim());
        } catch (IllegalArgumentException exception) {
            throw new IllegalStateException("JWT_SECRET deve estar codificada em Base64", exception);
        }
        if (chave.length < 32) {
            throw new IllegalStateException("JWT_SECRET deve possuir pelo menos 32 bytes");
        }
        return new SecretKeySpec(chave, "HmacSHA256");
    }

    @Bean
    JwtEncoder jwtEncoder(SecretKey secretKey) {
        return NimbusJwtEncoder.withSecretKey(secretKey)
                .algorithm(MacAlgorithm.HS256)
                .build();
    }

    @Bean
    JwtDecoder jwtDecoder(SecretKey secretKey, JwtProperties properties) {
        NimbusJwtDecoder decoder = NimbusJwtDecoder
                .withSecretKey(secretKey)
                .macAlgorithm(MacAlgorithm.HS256)
                .build();
        decoder.setJwtValidator(
                JwtValidators.createDefaultWithIssuer(properties.issuer()));
        return decoder;
    }
}

package com.curso.biblioteca2026.exception;

import jakarta.servlet.http.HttpServletRequest;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import java.time.Instant;
import java.util.LinkedHashMap;
import java.util.Map;

@RestControllerAdvice
public class ApiExceptionHandler {

    @ExceptionHandler(CredenciaisInvalidasException.class)
    public ResponseEntity<ApiError> tratarCredenciaisInvalidas(
            CredenciaisInvalidasException exception,
            HttpServletRequest request) {

        return construirResposta(
                HttpStatus.UNAUTHORIZED,
                exception.getMessage(),
                request.getRequestURI(),
                null
        );
    }

    @ExceptionHandler(RecursoNaoEncontradoException.class)
    public ResponseEntity<ApiError> tratarNaoEncontrado(
            RecursoNaoEncontradoException exception,
            HttpServletRequest request) {

        return construirResposta(
                HttpStatus.NOT_FOUND,
                exception.getMessage(),
                request.getRequestURI(),
                null
        );
    }

    @ExceptionHandler({RecursoDuplicadoException.class, RecursoEmUsoException.class})
    public ResponseEntity<ApiError> tratarConflito(
            RuntimeException exception,
            HttpServletRequest request) {

        return construirResposta(
                HttpStatus.CONFLICT,
                exception.getMessage(),
                request.getRequestURI(),
                null
        );
    }

    @ExceptionHandler({IllegalArgumentException.class, IllegalStateException.class, HttpMessageNotReadableException.class})
    public ResponseEntity<ApiError> tratarRequisicaoInvalida(
            Exception exception,
            HttpServletRequest request) {

        String mensagem = exception instanceof HttpMessageNotReadableException
                ? "RequisiÃ§Ã£o invÃ¡lida ou JSON malformado"
                : exception.getMessage();

        return construirResposta(
                HttpStatus.BAD_REQUEST,
                mensagem,
                request.getRequestURI(),
                null
        );
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiError> tratarValidacao(
            MethodArgumentNotValidException exception,
            HttpServletRequest request) {

        Map<String, String> campos = new LinkedHashMap<>();
        exception.getBindingResult()
                .getFieldErrors()
                .forEach(erro -> campos.put(erro.getField(), erro.getDefaultMessage()));

        return construirResposta(
                HttpStatus.BAD_REQUEST,
                "Dados invÃ¡lidos",
                request.getRequestURI(),
                campos
        );
    }

    private ResponseEntity<ApiError> construirResposta(
            HttpStatus status,
            String message,
            String path,
            Map<String, String> fields) {

        ApiError corpo = new ApiError(
                Instant.now(),
                status.value(),
                status.getReasonPhrase(),
                message,
                path,
                fields
        );

        return ResponseEntity.status(status).body(corpo);
    }
}

    @ElementCollection(fetch = FetchType.EAGER)
    @CollectionTable(
            name = "conta_acesso_perfil",
            joinColumns = @JoinColumn(
                    name = "conta_acesso_id",

                    foreignKey = @ForeignKey(name = "fk_conta_acesso_perfil_conta")
            )

    }

    public ContaAcesso(Pessoa pessoa, String login, String senhaHash) {
        this.pessoa = Objects.requireNonNull(pessoa, "Pessoa Ã© obrigatÃ³ria");
        this.login = validarLogin(login);
        this.senhaHash = validarHash(senhaHash);
        this.status = StatusContaAcesso.ATIVA;
        this.perfis.add(PerfilAcesso.OPERADOR);
    }

    }

        return pessoa;
    }

    }

        return senhaHash;
    }

        return status;
    }

    public Set<PerfilAcesso> getPerfis() {
        return Collections.unmodifiableSet(perfis);
    }

    public boolean podeAutenticar() {
        return this.status == StatusContaAcesso.ATIVA
                && this.pessoa != null
                && this.pessoa.getStatus() == Status.ATIVO;
    }

    public void alterarSenha(String novoSenhaHash) {
        this.senhaHash = validarHash(novoSenhaHash);
    }

    public void bloquear() {
        this.status = StatusContaAcesso.BLOQUEADA;
    }

    public void ativar() {
        this.status = StatusContaAcesso.ATIVA;
    }

    private static String validarLogin(String login) {
        if (login == null || login.isBlank()) {
            throw new IllegalArgumentException("Login Ã© obrigatÃ³rio");
        }
        String valor = login.trim();
        if (valor.length() < 3 || valor.length() > 80 || !valor.matches("^[A-Za-z0-9._-]+$")) {
            throw new IllegalArgumentException("Login invÃ¡lido");
        }
        return valor;

package com.curso.biblioteca2026.model;

public final class DocumentoFiscal {

    private DocumentoFiscal() {
    }

    public static boolean cpfValido(String cpf) {
        if (cpf == null || !cpf.matches("\\d{11}") || digitosRepetidos(cpf)) {
            return false;
        }
        int primeiro = calcularDigito(cpf.substring(0, 9), 10);
        int segundo = calcularDigito(cpf.substring(0, 9) + primeiro, 11);
        return cpf.equals(cpf.substring(0, 9) + primeiro + segundo);
    }

    public static boolean cnpjValido(String cnpj) {
        if (cnpj == null || !cnpj.matches("\\d{14}") || digitosRepetidos(cnpj)) {
            return false;
        }
        int primeiro = calcularDigitoCnpj(cnpj.substring(0, 12));
        int segundo = calcularDigitoCnpj(cnpj.substring(0, 12) + primeiro);
        return cnpj.equals(cnpj.substring(0, 12) + primeiro + segundo);
    }

    private static int calcularDigito(String base, int pesoInicial) {
        int soma = 0;
        for (int indice = 0; indice < base.length(); indice++) {
            soma += Character.getNumericValue(base.charAt(indice)) * (pesoInicial - indice);
        }
        int resto = soma % 11;
        return resto < 2 ? 0 : 11 - resto;
    }

    private static int calcularDigitoCnpj(String base) {
        int peso = base.length() - 7;
        int soma = 0;
        for (int indice = 0; indice < base.length(); indice++) {
            soma += Character.getNumericValue(base.charAt(indice)) * peso;
            peso--;
            if (peso == 1) {
                peso = 9;
            }
        }
        int resto = soma % 11;
        return resto < 2 ? 0 : 11 - resto;
    }

    private static boolean digitosRepetidos(String valor) {
        return valor.chars().allMatch(digito -> digito == valor.charAt(0));
    }
}

package com.curso.biblioteca2026.security;

import com.curso.biblioteca2026.config.JwtProperties;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.GrantedAuthority;
import org.springframework.security.oauth2.jwt.JwtClaimsSet;
import org.springframework.security.oauth2.jwt.JwtEncoder;
import org.springframework.security.oauth2.jwt.JwtEncoderParameters;
import org.springframework.stereotype.Service;

import java.time.Clock;
import java.time.Instant;
import java.util.List;
import java.util.UUID;

@Service
public class TokenService {

    private final JwtEncoder encoder;
    private final JwtProperties properties;
    private final Clock clock;

    public TokenService(JwtEncoder encoder, JwtProperties properties, Clock clock) {
        this.encoder = encoder;
        this.properties = properties;
        this.clock = clock;
    }

    public TokenAcesso emitir(Authentication authentication) {
        Instant emitidoEm = clock.instant();
        Instant expiraEm = emitidoEm.plus(properties.expiration());

        List<String> perfis = authentication.getAuthorities().stream()
                .map(GrantedAuthority::getAuthority)
                .filter(valor -> valor.startsWith("ROLE_"))
                .map(valor -> valor.substring("ROLE_".length()))
                .sorted()
                .toList();

        JwtClaimsSet claims = JwtClaimsSet.builder()
                .issuer(properties.issuer())
                .issuedAt(emitidoEm)
                .expiresAt(expiraEm)
                .subject(authentication.getName())
                .id(UUID.randomUUID().toString())
                .claim("perfis", perfis)
                .build();

        String valor = encoder
                .encode(JwtEncoderParameters.from(claims))
                .getTokenValue();

        return new TokenAcesso(valor, expiraEm, properties.expiration().toSeconds());
    }
}

                .sorted(Comparator.comparing(item -> item.getLivro().getId()))

        CategoriaLivro categoria = categoriaLivroService.buscarEntidade(request.categoriaId());

                request.quantidadeExemplares(),
                request.valorReposicao(),
                LocalDate.now(),
                request.ativo(),

                        .header(HttpHeaders.AUTHORIZATION, "Bearer " + token))

        String trocaSenhaJson = """

        mockMvc.perform(post("/api/autenticacao/login")
                        .contentType(MediaType.APPLICATION_JSON)
                        .content("""
                                {
                                  "login": "%s",
                                  "senha": "%s"
                                }
                                """.formatted(login, SENHA_INICIAL)))
                .andExpect(status().isUnauthorized());

        mockMvc.perform(get("/api/contas-acesso/me")
                        .header(HttpHeaders.AUTHORIZATION, "Bearer " + novoToken))
                .andExpect(status().isOk());

        ContaAcesso conta = contaAcessoRepository.findByLoginIgnoreCase(login).orElseThrow();
        conta.bloquear();
        contaAcessoRepository.saveAndFlush(conta);

        mockMvc.perform(post("/api/autenticacao/login")
                        .contentType(MediaType.APPLICATION_JSON)
                        .content("""
                                {
                                  "login": "%s",
                                  "senha": "%s"
                                }
                                """.formatted(login, NOVA_SENHA)))
                .andExpect(status().isUnauthorized());
    }

    private String autenticar(String login, String senha) throws Exception {
        String loginJson = """
                {
                  "login": "%s",
                  "senha": "%s"
                }
                """.formatted(login, senha);

        MvcResult result = mockMvc.perform(post("/api/autenticacao/login")
                        .contentType(MediaType.APPLICATION_JSON)
                        .content(loginJson))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.tokenType").value("Bearer"))
                .andExpect(jsonPath("$.expiresIn").value(1800))
                .andExpect(jsonPath("$.accessToken").isNotEmpty())
                .andReturn();

        Map<String, Object> body = JsonParserFactory.getJsonParser()
                .parseMap(result.getResponse().getContentAsString());
        return body.get("accessToken").toString();
    }

    private static String gerarCpfUnico() {
        long semente = Math.abs(System.nanoTime() + SEQUENCIA_CPF.incrementAndGet()) % 899_999_999L + 100_000_000L;
        String base = String.format("%09d", semente);
        final char primeiroDigito = base.charAt(0);
        if (base.chars().allMatch(c -> c == primeiroDigito)) {
            base = "123456789";
        }
        int d1 = calcularDigitoCpf(base, 10);
        int d2 = calcularDigitoCpf(base + d1, 11);
        return base + d1 + d2;
    }

    private static int calcularDigitoCpf(String base, int pesoInicial) {
        int soma = 0;
        for (int i = 0; i < base.length(); i++) {
            soma += Character.getNumericValue(base.charAt(i)) * (pesoInicial - i);
        }
        int resto = soma % 11;
        return resto < 2 ? 0 : 11 - resto;
    }

import com.curso.biblioteca2026.model.PerfilAcesso;
import com.curso.biblioteca2026.model.Pessoa;
import com.curso.biblioteca2026.model.StatusContaAcesso;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.test.context.ActiveProfiles;
import org.springframework.transaction.annotation.Transactional;

import java.util.Set;
import java.util.concurrent.atomic.AtomicInteger;

import static org.junit.jupiter.api.Assertions.*;

@SpringBootTest
@ActiveProfiles("test")
@Transactional
class ContaAcessoServiceTest {

    private static final AtomicInteger SEQUENCIA_CPF = new AtomicInteger(300_000_000);

    @Autowired
    private PessoaService pessoaService;

    @Autowired
    private ContaAcessoService contaAcessoService;

    @Autowired
    private PasswordEncoder passwordEncoder;

    @Test
    void devePersistirContaComHashBcryptSemGuardarSenhaAberta() {
        Pessoa pessoa = pessoaService.cadastrar(
                "Ana Souza",
                "ana." + System.nanoTime() + "@example.com",
                gerarCpfUnico());

        String senhaAberta = "uma-frase-senha-2026";
        String login = "ana." + System.nanoTime();

        ContaAcesso conta = contaAcessoService.cadastrar(pessoa.getId(), login, senhaAberta);

        assertNotNull(conta.getId());
        assertEquals(StatusContaAcesso.ATIVA, conta.getStatus());
        assertEquals(Set.of(PerfilAcesso.OPERADOR), conta.getPerfis());
        assertNotEquals(senhaAberta, conta.getSenhaHash());
        assertTrue(conta.getSenhaHash().startsWith("{bcrypt}"));
        assertTrue(passwordEncoder.matches(senhaAberta, conta.getSenhaHash()));
        assertFalse(conta.toString().contains(senhaAberta));
        assertFalse(conta.toString().contains(conta.getSenhaHash()));
    }

    @Test
    void deveRejeitarSenhaAcimaDe72BytesUtf8() {
        Pessoa pessoa = pessoaService.cadastrar(
                "Carlos Mendes",
                "carlos." + System.nanoTime() + "@example.com",
                gerarCpfUnico());

        String senha73Bytes = "Ã¡".repeat(36) + "a"; // 36 * 2 + 1 = 73 bytes em UTF-8

        assertThrows(
                IllegalArgumentException.class,
                () -> contaAcessoService.cadastrar(pessoa.getId(), "carlos." + System.nanoTime(), senha73Bytes));
    }

    private static String gerarCpfUnico() {
        long semente = Math.abs(System.nanoTime() + SEQUENCIA_CPF.incrementAndGet()) % 899_999_999L + 100_000_000L;
        String base = String.format("%09d", semente);
        final char primeiroDigito = base.charAt(0);
        if (base.chars().allMatch(c -> c == primeiroDigito)) {
            base = "123456789";
        }
        int d1 = calcularDigitoCpf(base, 10);
        int d2 = calcularDigitoCpf(base + d1, 11);
        return base + d1 + d2;
    }

    private static int calcularDigitoCpf(String base, int pesoInicial) {
        int soma = 0;
        for (int i = 0; i < base.length(); i++) {
            soma += Character.getNumericValue(base.charAt(i)) * (pesoInicial - i);
        }
        int resto = soma % 11;
        return resto < 2 ? 0 : 11 - resto;
    }
}

    private static final AtomicInteger SEQUENCIA_CPF = new AtomicInteger(100_000_000);

    @Autowired

    private PessoaService pessoaService;

    private LeitorService leitorService;

    @Test
    void deveFazerRollbackQuandoSegundoItemNaoPossuiExemplaresSuficientes() {
        CategoriaLivro categoria = categoriaRepository.save(new CategoriaLivro("FicÃ§Ã£o Rollback " + System.nanoTime()));
        Livro livro1 = livroRepository.save(new Livro(
                "ISBN1-" + System.nanoTime(), "Livro Um", 5, new BigDecimal("30.00"), LocalDate.now(), true, categoria));
        Livro livro2 = livroRepository.save(new Livro(
                "ISBN2-" + System.nanoTime(), "Livro Dois", 1, new BigDecimal("40.00"), LocalDate.now(), true, categoria));

        Participantes participantes = criarParticipantes();

        Emprestimo emprestimo = emprestimoService.cadastrar(participantes.leitor().getId(), participantes.bibliotecario().getId());
        emprestimoService.adicionarItem(emprestimo.getId(), livro1.getId(), 2);
        emprestimoService.adicionarItem(emprestimo.getId(), livro2.getId(), 3);

        assertThrows(IllegalArgumentException.class, () -> emprestimoService.finalizar(emprestimo.getId()));

        Livro livro1Banco = livroRepository.findById(livro1.getId()).orElseThrow();
        Emprestimo emprestimoBanco = emprestimoService.buscarPorId(emprestimo.getId());

        assertEquals(5, livro1Banco.getQuantidadeExemplares());
        assertEquals(StatusEmprestimo.ABERTO, emprestimoBanco.getStatus());
    }

    @Test
    void deveImpedirEmprestimoAcimaDoSaldoEmFinalizacoesConcorrentes() throws Exception {
        CategoriaLivro categoria = categoriaRepository.save(new CategoriaLivro("FicÃ§Ã£o Concorrente " + System.nanoTime()));
        Livro livro = livroRepository.save(new Livro(
                "CONC-" + System.nanoTime(), "Livro Disputado", 5, new BigDecimal("50.00"), LocalDate.now(), true, categoria));

        Participantes participantes = criarParticipantes();
        Emprestimo primeiro = criarEmprestimoComItem(participantes, livro, 4);
        Emprestimo segundo = criarEmprestimoComItem(participantes, livro, 4);

        CountDownLatch inicioSimultaneo = new CountDownLatch(1);

        try (ExecutorService executor = Executors.newFixedThreadPool(2)) {
            Future<ResultadoFinalizacao> primeiroResultado = executor.submit(
                    () -> finalizarAposSinal(primeiro.getId(), inicioSimultaneo));
            Future<ResultadoFinalizacao> segundoResultado = executor.submit(
                    () -> finalizarAposSinal(segundo.getId(), inicioSimultaneo));

            inicioSimultaneo.countDown();

            List<ResultadoFinalizacao> resultados = List.of(
                    primeiroResultado.get(10, TimeUnit.SECONDS),
                    segundoResultado.get(10, TimeUnit.SECONDS));

            assertEquals(1, resultados.stream().filter(ResultadoFinalizacao::sucesso).count());
            assertEquals(1, resultados.stream().filter(r -> !r.sucesso()).count());
            assertTrue(resultados.stream()
                    .filter(r -> !r.sucesso())
                    .allMatch(r -> "Quantidade de exemplares insuficiente para retirada".equals(r.mensagem())));
        }

        assertEquals(1, livroRepository.findById(livro.getId()).orElseThrow().getQuantidadeExemplares());

        List<StatusEmprestimo> estados = List.of(
                emprestimoService.buscarPorId(primeiro.getId()).getStatus(),
                emprestimoService.buscarPorId(segundo.getId()).getStatus());
        assertEquals(1, estados.stream().filter(StatusEmprestimo.FINALIZADO::equals).count());
        assertEquals(1, estados.stream().filter(StatusEmprestimo.ABERTO::equals).count());
    }

    private Participantes criarParticipantes() {
        Pessoa pessoa = pessoaService.cadastrar(
                "Carlos Silva",
                "carlos." + System.nanoTime() + "@example.com",
                gerarCpfUnico());
        Leitor leitor = leitorService.cadastrar(pessoa.getId(), "(17) 98888-7777");
        Bibliotecario bib = bibliotecarioService.cadastrar(
                pessoa.getId(),
                "MAT-" + System.nanoTime(),
                FuncaoBibliotecario.ATENDENTE,
                LocalDate.now());
        return new Participantes(leitor, bib);
    }

    private static String gerarCpfUnico() {
        long semente = Math.abs(System.nanoTime() + SEQUENCIA_CPF.incrementAndGet()) % 899_999_999L + 100_000_000L;
        String base = String.format("%09d", semente);
        final char primeiroDigito = base.charAt(0);
        if (base.chars().allMatch(c -> c == primeiroDigito)) {
            base = "123456789";
        }
        int d1 = calcularDigitoCpf(base, 10);
        int d2 = calcularDigitoCpf(base + d1, 11);
        return base + d1 + d2;
    }

    private static int calcularDigitoCpf(String base, int pesoInicial) {
        int soma = 0;
        for (int i = 0; i < base.length(); i++) {
            soma += Character.getNumericValue(base.charAt(i)) * (pesoInicial - i);
        }
        int resto = soma % 11;
        return resto < 2 ? 0 : 11 - resto;
    }

    private Emprestimo criarEmprestimoComItem(Participantes participantes, Livro livro, int quantidade) {
        Emprestimo emprestimo = emprestimoService.cadastrar(
                participantes.leitor().getId(),
                participantes.bibliotecario().getId());
        emprestimoService.adicionarItem(emprestimo.getId(), livro.getId(), quantidade);
        return emprestimo;
    }

    private ResultadoFinalizacao finalizarAposSinal(Long emprestimoId, CountDownLatch inicioSimultaneo)
            throws InterruptedException {
        inicioSimultaneo.await();
        try {
            emprestimoService.finalizar(emprestimoId);
            return new ResultadoFinalizacao(true, null);
        } catch (IllegalArgumentException exception) {
            return new ResultadoFinalizacao(false, exception.getMessage());
        }
    }

    private record Participantes(Leitor leitor, Bibliotecario bibliotecario) {
    }

    }
